# ---------------------------------------------------------------------------
# Logging (simple file + console logger)
# ---------------------------------------------------------------------------
def log(msg: str, level: str = "INFO"):
    ts = datetime.now().strftime("%H:%M:%S")
    line = f"[{ts}] [{level}] {msg}"
    print(line)
    with open(LOG_FILE, "a") as f:
        f.write(line + "\n")

# Workspace helpers
# ---------------------------------------------------------------------------
def ensure_workspace():
    os.makedirs(WORKSPACE_DIR, exist_ok=True)
    os.makedirs(os.path.join(WORKSPACE_DIR, "reviews"), exist_ok=True)


def list_workspace(path: str = "") -> str:
    """List files in workspace, optionally under a subpath."""
    target = os.path.join(WORKSPACE_DIR, path) if path else WORKSPACE_DIR
    if not os.path.exists(target):
        return f"Path not found: {path}"
    if os.path.isfile(target):
        with open(target, "r", encoding="utf-8", errors="replace") as f:
            return f.read()[:8000]  # Cap file reads
    items = []
    for root, dirs, files in os.walk(target):
        rel = os.path.relpath(root, WORKSPACE_DIR)
        depth = rel.count(os.sep)
        if depth > 2:
            dirs.clear()
            continue
        for fname in files:
            items.append(os.path.join(rel, fname) if rel != "." else fname)
    return "\n".join(sorted(items)) if items else "(empty directory)"


# ---------------------------------------------------------------------------
# State tracking (replaces your AgentContext token tracking)
# ---------------------------------------------------------------------------
class SessionState:
    """Tracks dispatches, checkpoints, and worker results."""

    def __init__(self):
        self.step = 0
        self.dispatches: list[dict] = []
        self.worker_results: list[dict] = []

    def save_checkpoint(self):
        checkpoint = {
            "timestamp": datetime.now().isoformat(),
            "step": self.step,
            "dispatches": self.dispatches,
            "worker_count": len(self.worker_results),
        }
        with open("checkpoint.json", "w") as f:
            json.dump(checkpoint, f, indent=2, default=str)
        log(f"Checkpoint saved at step {self.step}")


state = SessionState()

# Human approval gate (same logic as your original)
# ---------------------------------------------------------------------------
def human_approval_gate(target: str, payload: str) -> str | None:
    if not HUMAN_APPROVAL_GATE:
        return None

    print(f"\n{'='*60}")
    print(f"  DISPATCH PENDING: [{target}]")
    print(f"  Instructions: {payload[:200]}")
    print(f"{'='*60}")

    try:
        user_input = input("ENTER to approve, or type override: ").strip()
    except (EOFError, KeyboardInterrupt):
        print("\n[Interrupted]")
        return "__ABORT__"

    return user_input if user_input else None

async def run_worker(agent_name: str, instructions: str) -> str:
    """
    Execute a worker agent using the SDK's query().

    This replaces your execute_worker() + AgentContext.chat() loop.
    The SDK handles the entire agent loop — tool calls, retries, context —
    and returns structured results.
    """
    config = WORKER_CONFIGS.get(agent_name)
    if not config:
        return json.dumps({"status": "error", "summary": f"Unknown agent: {agent_name}"})

    log(f"Dispatching to [{agent_name}]")
    print(f"\n--> DISPATCHING to [{agent_name}]...")

    result_text = ""
    try:
        async for message in query(
            prompt=instructions,
            options=ClaudeAgentOptions(
                system_prompt=config["system_prompt"],
                allowed_tools=config["allowed_tools"],
                max_turns=25,  # Workers get limited turns
                cwd=WORKSPACE_DIR,  # Workers operate in the project directory
            ),
        ):
            # Collect the final result message
            if hasattr(message, "result") and message.result:
                result_text = message.result
            # Optionally log intermediate assistant messages
            elif hasattr(message, "content"):
                for block in getattr(message, "content", []):
                    if hasattr(block, "text") and block.text:
                        # Grab the last substantive text as fallback
                        result_text = block.text

    except Exception as e:
        log(f"Worker {agent_name} failed: {e}", "ERROR")
        result_text = json.dumps({
            "status": "failed",
            "summary": f"Worker error: {type(e).__name__}: {str(e)}",
            "files_modified": [],
        })

    # Build structured result
    result = {
        "status": "completed",
        "agent": agent_name,
        "summary": result_text[:2000] if result_text else "No output captured",
    }

    state.worker_results.append(result)
    state.save_checkpoint()

    output = json.dumps(result, indent=2)
    print(f"<-- [{agent_name}] done ({len(output)} chars)")
    return output


# ---------------------------------------------------------------------------
# Custom MCP tools for the Lead Agent
#
# These replace your manage_workflow and inspect_workspace tool definitions.
# The @tool decorator creates in-process MCP tools — no subprocess needed.
# ---------------------------------------------------------------------------

@tool(
    "dispatch_to_agent",
    "Delegate a task to a specialist worker agent. Available agents: "
    "frontend_agent, security_agent, compliance_agent, research_agent. "
    "Provide detailed instructions for the agent.",
    {"agent_name": str, "instructions": str},
)
async def dispatch_to_agent(args: dict) -> dict:
    agent_name = args["agent_name"]
    instructions = args["instructions"]

    # Validate agent
    if agent_name not in WORKER_CONFIGS:
        return {
            "content": [{"type": "text", "text": (
                f"ERROR: Unknown agent '{agent_name}'. "
                f"Available: {list(WORKER_CONFIGS.keys())}"
            )}]
        }

    # Human approval gate
    override = human_approval_gate(agent_name, instructions)
    if override == "__ABORT__":
        return {"content": [{"type": "text", "text": "Session aborted by user."}]}
    if override:
        log(f"Human override: {override[:100]}")
        return {"content": [{"type": "text", "text": (
            f"HOLD: User intervened with new instructions: {override}"
        )}]}

    # Track dispatch
    state.step += 1
    state.dispatches.append({
        "step": state.step,
        "agent": agent_name,
        "instructions": instructions[:200],
        "timestamp": datetime.now().isoformat(),
    })

    # Run the worker
    result = await run_worker(agent_name, instructions)

    return {"content": [{"type": "text", "text": f"Output from {agent_name}:\n{result}"}]}


@tool(
    "inspect_workspace",
    "List files or read a specific file from the project workspace. "
    "Use to verify worker output after dispatches. "
    "Pass empty path to list all files, or a relative path to read a file.",
    {"path": str},
)
async def inspect_workspace_tool(args: dict) -> dict:
    path = args.get("path", "")
    result = list_workspace(path)
    return {"content": [{"type": "text", "text": result}]}


@tool(
    "log_comment",
    "Log an observation or intermediate thought. Use between dispatches "
    "to record your reasoning.",
    {"comment": str},
)
async def log_comment(args: dict) -> dict:
    comment = args["comment"]
    print(f"\n[LEAD COMMENT]: {comment}")
    log(f"Lead comment: {comment[:200]}")
    return {"content": [{"type": "text", "text": "Comment logged. Continue."}]}


@tool(
    "finish_project",
    "Declare the project complete. Provide a final summary of what was accomplished.",
    {"summary": str},
)
async def finish_project(args: dict) -> dict:
    summary = args["summary"]
    print(f"\n[PROJECT COMPLETE]: {summary}")
    log(f"Project complete: {summary[:500]}")
    state.save_checkpoint()
    return {"content": [{"type": "text", "text": "Project marked complete. Session ending."}]}

LEAD_SYSTEM_PROMPT = (
    "You are the Orchestrator for Financial GPS, a consumer-facing financial "
    "planning web application. You hold the Master Plan (guide.md) and coordinate "
    "a team of 4 specialist agents to execute it.\n\n"
    "YOUR TEAM:\n"
    "- frontend_agent: Frontend Architect. Writes code, creates components, implements UI.\n"
    "- security_agent: Security Analyst. Reviews code for vulnerabilities.\n"
    "- compliance_agent: Compliance Reviewer. Reviews content for regulatory compliance.\n"
    "- research_agent: Web Researcher. Finds information, verifies facts.\n\n"
    "MATH PROHIBITION:\n"
    "NEVER perform arithmetic internally. Dispatch to frontend_agent for any calculations.\n\n"
    "YOUR TOOLS:\n"
    "- dispatch_to_agent: Delegate work to a specialist. Your PRIMARY tool.\n"
    "- inspect_workspace: Verify worker output by listing/reading project files.\n"
    "- log_comment: Record observations between dispatches.\n"
    "- finish_project: Declare the project complete.\n\n"
    "WORKFLOW:\n"
    "1. Read the next step from the Master Plan\n"
    "2. Dispatch to the designated agent with detailed instructions\n"
    "3. Inspect workspace to verify output quality\n"
    "4. If security/compliance review needed, dispatch to those agents\n"
    "5. Log observations, then proceed to next step\n"
    "6. Repeat until all plan steps complete, then finish.\n\n"
    "Follow the Master Plan step by step. Each step specifies which agent to use. "
    "The plan has 21 steps in 3 sub-phases (1A, 1B, 1C) plus final integration."
)


async def run_orchestrator():
    # --- 0. Prepare ---
    ensure_workspace()
    files = os.listdir(WORKSPACE_DIR)
    print(f"[PROJECT] Working directory: {WORKSPACE_DIR} ({len(files)} items)")

    # --- 1. Load the Guide ---
    guide_path = "guide.md"
    if not os.path.exists(guide_path):
        print(f"ERROR: Create '{guide_path}' with your project plan before running.")
        return

    with open(guide_path, "r", encoding="utf-8") as f:
        guide = f.read()

    # --- 2. Build MCP server with custom tools ---
    lead_tools_server = create_sdk_mcp_server(
        name="orchestrator-tools",
        version="1.0.0",
        tools=[dispatch_to_agent, inspect_workspace_tool, log_comment, finish_project],
    )

    # --- 3. Run the Lead Agent ---
    print("--- SYSTEM ONLINE ---")
    print("[TEAM] frontend_agent | security_agent | compliance_agent | research_agent")
    mode = "Human approval required" if HUMAN_APPROVAL_GATE else "Fully autonomous"
    print(f"[MODE] {mode}")
    log("Orchestrator started")

    prompt = f"Here is the Master Plan:\n\n{guide}\n\nBegin execution of Step 1."

    options = ClaudeAgentOptions(
        system_prompt=LEAD_SYSTEM_PROMPT,
        # Only allow our custom MCP tools — lead should NOT write files directly
        allowed_tools=[
            "mcp__orchestrator-tools__dispatch_to_agent",
            "mcp__orchestrator-tools__inspect_workspace",
            "mcp__orchestrator-tools__log_comment",
            "mcp__orchestrator-tools__finish_project",
        ],
        mcp_servers={"orchestrator-tools": lead_tools_server},
        max_turns=MAX_LEAD_TURNS,
        cwd=WORKSPACE_DIR,
    )

    try:
        async with ClaudeSDKClient(options=options) as client:
            await client.query(prompt)
            async for message in client.receive_response():
                # The SDK streams messages as the lead agent works.
                # Tool calls (dispatches) are handled by our custom tools above.
                # We just log any text output from the lead.
                if hasattr(message, "result") and message.result:
                    print(f"\n[LEAD RESULT]: {message.result[:500]}")
                elif hasattr(message, "content"):
                    for block in getattr(message, "content", []):
                        if hasattr(block, "text") and block.text:
                            text = block.text.strip()
                            if text:
                                print(f"\n[LEAD]: {text[:300]}")
    except KeyboardInterrupt:
        print("\n--- SESSION INTERRUPTED ---")
    except Exception as e:
        log(f"Orchestrator failed: {e}", "ERROR")
        print(f"\n[FATAL] {e}")

    # --- 4. Final Report ---
    state.save_checkpoint()
    print(f"\n--- SESSION SUMMARY ---")
    print(f"Total dispatches: {len(state.dispatches)}")
    print(f"Workers completed: {len(state.worker_results)}")
    for d in state.dispatches:
        print(f"  Step {d['step']}: {d['agent']} — {d['instructions'][:80]}")
    log("Session complete")
